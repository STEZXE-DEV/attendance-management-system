# =========================
# USERS API (PowerShell - Invoke-RestMethod)
# =========================

# GET ALL USERS
Invoke-RestMethod -Uri "http://127.0.0.1:5000/users" -Method GET

# GET USER BY ID
Invoke-RestMethod -Uri "http://127.0.0.1:5000/users/1" -Method GET

# CREATE USER
Invoke-RestMethod -Uri "http://127.0.0.1:5000/users" -Method POST -ContentType "application/json" -Body '{"name":"Piotr","surname":"Zielinski","email":"piotr@test.com","password":"1234","role":"teacher"}'

# UPDATE USER
Invoke-RestMethod -Uri "http://127.0.0.1:5000/users/1" -Method PUT -ContentType "application/json" -Body '{"name":"PiotrUpdated","surname":"Zielinski","email":"piotr@test.com","role":"teacher"}'

# DELETE USER
Invoke-RestMethod -Uri "http://127.0.0.1:5000/users/1" -Method DELETE
